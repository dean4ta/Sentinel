This is the c++ code used for the ESP32 that controlled the driving. The ESP should be 
connected to the Orin through serial. Then the PWM signals to the mixer are sent through
the GPIO pins defined here. The LEDs data pin are also connected to one of the ESP's GPIO
pins. A PCB design may have been provided, but as of the time of this document being written
there is none. 

// ESP32 Serial-to-PWM Motor Controller
//
// Purpose:
// This program receives cleaned movement commands from the Jetson Orin
// over USB serial and converts them into 50 Hz servo-style PWM signals
// for the goBILDA mixer board.
//
// It also receives LED state commands from the Orin and updates the
// Sentinel 1 LED strip to show the robot's current mode.
//
// Expected motor command format:
// drive,rotate,strafe\n
//
// Example:
// 0.25,-0.10,0.50
//
// Expected state command format:
// STATE,BOOT
// STATE,READY
// STATE,SAFE
// STATE,FAULT
//
// The Orin is responsible for joystick processing and deadzone filtering.
// The ESP32 is responsible only for PWM output, LED output, validation,
// direction correction, rotation limiting, and failsafe behavior.

#include <Adafruit_NeoPixel.h>

// -------------------- LED strip settings --------------------

const int LED_PIN = 18;
const int NUM_LEDS = 30;

Adafruit_NeoPixel strip(NUM_LEDS, LED_PIN, NEO_GRB + NEO_KHZ800);

// -------------------- PWM output pins --------------------

const int DRIVE_PIN  = 23;
const int ROTATE_PIN = 22;
const int STRAFE_PIN = 21;

// -------------------- PWM signal settings --------------------

const int PWM_FREQ = 50;
const int PWM_RES  = 16;

const int MIN_US     = 1000;
const int NEUTRAL_US = 1500;
const int MAX_US     = 2000;

// -------------------- Movement adjustments --------------------

const bool INVERT_STRAFE = true;
const float ROTATE_LIMIT = 0.5;

// -------------------- Serial and failsafe settings --------------------

const unsigned long FAILSAFE_MS = 200;
unsigned long lastMotorCommandTime = 0;

String currentLine = "";
String newestCompleteLine = "";

bool failsafeActive = false;

// -------------------- PWM helper functions --------------------

int pulseToDuty(int pulse_us) {
  const int period_us = 1000000 / PWM_FREQ;
  const int maxDuty = (1 << PWM_RES) - 1;

  pulse_us = constrain(pulse_us, MIN_US, MAX_US);

  return (int)((pulse_us / (float)period_us) * maxDuty);
}

int commandToPulse(float value) {
  value = constrain(value, -1.0, 1.0);

  int pulse = NEUTRAL_US + (int)(value * 500.0);

  return constrain(pulse, MIN_US, MAX_US);
}

void writePWM(int pin, int pulse_us) {
  ledcWrite(pin, pulseToDuty(pulse_us));
}

void neutralAll() {
  writePWM(DRIVE_PIN, NEUTRAL_US);
  writePWM(ROTATE_PIN, NEUTRAL_US);
  writePWM(STRAFE_PIN, NEUTRAL_US);
}

// -------------------- LED helper functions --------------------

void setLEDColor(uint8_t r, uint8_t g, uint8_t b) {
  for (int i = 0; i < NUM_LEDS; i++) {
    strip.setPixelColor(i, strip.Color(r, g, b));
  }

  strip.show();
}

void handleStateCommand(String line) {
  line.trim();

  if (line == "STATE,BOOT") {
    setLEDColor(255, 255, 255);   // white, boot or standby
  }
  else if (line == "STATE,READY") {
    setLEDColor(0, 0, 255);       // blue, manual drive / ready
  }
  else if (line == "STATE,SAFE") {
    setLEDColor(0, 255, 0);       // green, safe drive
  }
  else if (line == "STATE,FAULT") {
    setLEDColor(255, 0, 0);       // red, fault
  }
}

// -------------------- Command validation --------------------

bool isValidNumber(String text) {
  text.trim();

  if (text.length() == 0) {
    return false;
  }

  bool decimalSeen = false;
  bool digitSeen = false;

  for (int i = 0; i < text.length(); i++) {
    char c = text.charAt(i);

    if (c >= '0' && c <= '9') {
      digitSeen = true;
    }
    else if (c == '.' && !decimalSeen) {
      decimalSeen = true;
    }
    else if (c == '-' && i == 0) {
      // Negative sign is allowed only at the start
    }
    else {
      return false;
    }
  }

  return digitSeen;
}

// -------------------- Command parsing --------------------

bool parseAndApplyCommand(String line) {
  line.trim();

  if (line.length() == 0) {
    return false;
  }

  // LED state commands do not count as motor commands.
  // This prevents LED messages from keeping the drive system alive
  // if motor commands stop.
  if (line.startsWith("STATE")) {
    handleStateCommand(line);
    return false;
  }

  int firstComma = line.indexOf(',');
  int secondComma = line.indexOf(',', firstComma + 1);

  if (firstComma == -1 || secondComma == -1) {
    return false;
  }

  String driveText  = line.substring(0, firstComma);
  String rotateText = line.substring(firstComma + 1, secondComma);
  String strafeText = line.substring(secondComma + 1);

  if (!isValidNumber(driveText) || !isValidNumber(rotateText) || !isValidNumber(strafeText)) {
    return false;
  }

  float drive  = driveText.toFloat();
  float rotate = rotateText.toFloat();
  float strafe = strafeText.toFloat();

  // Reject obviously bad values instead of trusting them.
  // Slightly out-of-range values are still clamped later, but values this far
  // out usually mean a corrupted serial line or a software bug upstream.
  if (drive < -1.2 || drive > 1.2 ||
      rotate < -1.2 || rotate > 1.2 ||
      strafe < -1.2 || strafe > 1.2) {
    return false;
  }

  // Correct strafe direction.
  if (INVERT_STRAFE) {
    strafe = -strafe;
  }

  // Limit rotation to half speed.
  rotate = constrain(rotate, -ROTATE_LIMIT, ROTATE_LIMIT);

  writePWM(DRIVE_PIN, commandToPulse(drive));
  writePWM(ROTATE_PIN, commandToPulse(rotate));
  writePWM(STRAFE_PIN, commandToPulse(strafe));

  lastMotorCommandTime = millis();
  failsafeActive = false;

  return true;
}

// -------------------- Serial reading --------------------

void readSerialNewestOnly() {
  while (Serial.available() > 0) {
    char c = Serial.read();

    if (c == '\n') {
      newestCompleteLine = currentLine;
      currentLine = "";
    }
    else if (c != '\r') {
      currentLine += c;

      // Prevent corrupted or oversized serial lines from growing forever.
      if (currentLine.length() > 80) {
        currentLine = "";
      }
    }
  }

  // Only apply the newest complete command.
  // This prevents old joystick commands from building up and causing lag.
  if (newestCompleteLine.length() > 0) {
    parseAndApplyCommand(newestCompleteLine);
    newestCompleteLine = "";
  }
}

// -------------------- Setup --------------------

void setup() {
  Serial.begin(460800);

  strip.begin();
  strip.show();
  setLEDColor(255, 255, 255);   // white, boot or standby

  ledcAttach(DRIVE_PIN, PWM_FREQ, PWM_RES);
  ledcAttach(ROTATE_PIN, PWM_FREQ, PWM_RES);
  ledcAttach(STRAFE_PIN, PWM_FREQ, PWM_RES);

  neutralAll();

  lastMotorCommandTime = millis();
  failsafeActive = false;

  Serial.println("ESP32 motor controller ready");
}

// -------------------- Main loop --------------------

void loop() {
  readSerialNewestOnly();

  if (millis() - lastMotorCommandTime > FAILSAFE_MS) {
    if (!failsafeActive) {
      neutralAll();
      failsafeActive = true;
    }
  }
}
